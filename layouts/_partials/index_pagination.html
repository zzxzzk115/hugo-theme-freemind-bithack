{{- /* Receives a Paginator. Numbered pages: first, last, and two either side of the current one; gaps become "…". */ -}} {{- if gt .TotalPages 1 }} {{- $cur := .PageNumber }}{{ $total := .TotalPages }}
{{- if .HasPrev }} {{ T "prev_page" }} {{- else }} {{ T "prev_page" }} {{- end }} {{- range .Pagers }} {{- $n := .PageNumber }} {{- $d := sub $n $cur }} {{- if or (eq $n 1) (eq $n $total) (and (ge $d -2) (le $d 2)) }} {{- if eq $n $cur }} {{ $n }} {{- else }} {{ $n }} {{- end }} {{- else if or (eq $d -3) (eq $d 3) }} … {{- end }} {{- end }} {{- if .HasNext }} {{ T "next_page" }} {{- else }} {{ T "next_page" }} {{- end }}
{{- end }}
